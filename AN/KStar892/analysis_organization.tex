		\item[Data set and event selection]~\\
         The following datasets were used for this analysis
         \begin{itemize}
            \item \texttt{Run14 200 GeV \HeAu pro102 (MinBias)}: 2,838,783,497 events across 133 runs
            \item \texttt{Run15 200 GeV \pAl pro104 (MinBias)}: 2,036,978,459 events across 151 runs 
            \item \texttt{Run15 200 GeV \pAu CA (No VTX) pro104 (MinBias)}: 3,773,931,389 events across 417 runs
         \end{itemize}
         Each run was cheked for consistency of multiplicity and DC board distributions, Sec. \ref{sec:run_selection}
		\item[Analysis cuts]~\\
			The following data selection criteria were implemented:
			\begin{itemize}
            \item \textbf{Event selection}
					\begin{itemize}
                  \item Trigger cut: BBCLL1($>0$ tubes) or BBCLL1($>0$ tubes) narrowvtx
                  \item BBCz vertex cut: $\lvert Z_{\text{vertex}} \rvert < 30$
                  \item Pile-up rejection cut using $\text{frac}$ variable (\href{https://phenix-intra.sdcc.bnl.gov/phenix/WWW/p/draft/dcm07e/DoubleInteractionAN/small-system-double.pdf}{AN1304}), Sec. \ref{sec:pile_up_rejection}
					\end{itemize}
            \item \textbf{Charged tracks selection} \\					
					\label{text:track_selection}
					The following cuts were applied for every track
					\begin{itemize}
						\item Track quality cut: 63 or 31 or 51
                  \item DC zed cut: $3 < \lvert Z_{\text{DC}} \rvert < 75$
                  \item Momentum cut on charged tracks: $0.3 < p_\text{T} < 10$ GeV/c
                  \item DC fiducial cuts: 2D cuts in $\text{board}_{\text{DC}} - \alpha_{\text{DC}}$ space (Fig. \ref{fig:dm_dce0} - \ref{fig:dm_dcw1})
                  \item PC1 fiducial cuts: 2D cuts in $Z_{\text{PC1}} - \alpha_{\text{PC1}}$ space (Fig. \ref{fig:dm_pc1e} - \ref{fig:dm_pc1w})
						\item Track matching: $3\sigma$ on both $z$ and $\varphi$ residuals in at least one of the following detectors: PC3, TOFe, TOFw, EMCal
					\end{itemize}

            \item \textbf{Detectors specific cuts} \\					
					Additional detector specific cuts for the track to be associated with the given detector are described below

               PC2 (Run15 \pAl and Run15 \pAu only):
					\begin{itemize}
                  \item PC2 fiducial cuts: 2D cuts in $Z_{\text{PC2}} - \varphi_{\text{PC2}}$ space (Fig. \ref{fig:dm_pc2e})
					\end{itemize}

					PC3:
					\begin{itemize}
                  \item PC3 fiducial cuts: 2D cuts in $Z_{\text{PC3}} - \varphi_{\text{PC3}}$ space (Fig. \ref{fig:dm_pc3e} - \ref{fig:dm_pc3w})
					\end{itemize}
					
					TOFe:
					\begin{itemize}
                  \item TOFe fiducial cuts: 2D cuts in $\text{chamber}_{\text{TOF}} - \text{slat}_{\text{TOF}}$ space (Fig. \ref{fig:dm_tofw})
                  \item $\beta$ dependent $E_{\text{loss}}$ cut, Sec. \ref{sec:tofe_eloss_cut} (similar to \href{https://phenix-intra.sdcc.bnl.gov/phenix/WWW/p/draft/chujo/PPG026/AnalysisNote/}{AN187})
                  \item TOFw cuts on slats with bad $t - t_{exp}^{\pi^\pm}$ distributions (Sec. \ref{sec:tof_additional_cailbrations})
					\end{itemize}

					TOFw:
					\begin{itemize}
                  \item TOFw fiducial cuts: 2D cuts in $\text{chamber}_{\text{TOFw}} - \text{strip}_{\text{TOFw}}$ space (Fig. \ref{fig:dm_tofe})
                  \item ADC charge cut: $60 < q_{\text{TOFw}} < 600$, Sec. \ref{sec:tofw_adc_cut} (\href{https://phenix-intra.sdcc.bnl.gov/phenix/WWW/p/draft/belmonrj/AnalysisNotes/ananoterun7.pdf}{AN814})
                  \item TOFw cuts on strips with bad $t - t_{exp}^{\pi^\pm}$ distributions (Sec. \ref{sec:tof_additional_cailbrations})
					\end{itemize}

					EMCal:
					\begin{itemize}
                  \item EMCal fiducial cuts: 2D cuts in $Y_{\text{tower}} - Z_{\text{tower}}$ space (Fig. \ref{fig:dm_emcale0} - \ref{fig:dm_emcalw3})
                  \item $E_{core}$ cut: $E_{\text{core}} > 0.25$ for PbSc, $E_{\text{core}} > 0.35$ for PbGl, Sec. \ref{sec:emcal_ecore_cut} (similar to \href{https://phenix-intra.sdcc.bnl.gov/phenix/WWW/p/info/an/375/}{AN375})
					\end{itemize}

            \item \textbf{Track pair selection}
					\begin{itemize}
                  \item DC-PC1 ghost rejection cut (\href{https://phenix-intra.sdcc.bnl.gov/phenix/WWW/p/draft/nihashi/main.pdf}{AN1099}, \href{https://phenix-intra.sdcc.bnl.gov/phenix/WWW/p/draft/chujo/PPG101/AN_pp_paper/draft_AN.pdf}{AN776})
						\item Same arm cut: both tracks must lie in the same arm
                  \item Ghost rejection cut for PC2 (Run15 \pAl, Run15 \pAu only), PC3, TOFe, TOFw, EMCal, Sec. \ref{sec:ghost_cuts}
					\end{itemize}
			\end{itemize}
